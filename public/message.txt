repeat task.wait(5) until game:IsLoaded()

_G.ConfigGame = {
    Dashboard = {
        Enable = true,
        Url = "https://web-tracker2.vercel.app/roblox/update",
        Key = "a49969d795ad4e0fbc2962042230e0c8",
        Interval = 5,
    },
}

local Players = game:GetService("Players")
local HttpService = game:GetService("HttpService")

local player = Players.LocalPlayer
repeat task.wait(.2) until player

local lastDashboardSend = 0

-- ฟังก์ชันดึงเหรียญทอง (leaderstats) เป็นค่าพื้นฐานที่ส่งทุกแมพ
local function getCoins()
    local stats = player:FindFirstChild("leaderstats")
    local coins = stats and stats:FindFirstChild("Coins")

    if not coins then
        return 0
    end

    return tonumber(coins.Value) or 0
end

-- ฟังก์ชันหลักในการส่งข้อมูลไปยัง Dashboard หลังบ้าน
local function SendDashboardUpdate(statusText, itemsArray)
    local cfg = _G.ConfigGame and _G.ConfigGame.Dashboard
    if not cfg or not cfg.Enable then return end
    if not cfg.Url or cfg.Url == "" then return end

    if os.clock() - lastDashboardSend < (cfg.Interval or 5) then
        return
    end
    lastDashboardSend = os.clock()

    -- ตรวจสอบ Executor Request Function
    local requestFunc =
        (syn and syn.request)
        or (http and http.request)
        or http_request
        or request

    if not requestFunc then
        warn("Dashboard request function not found")
        return
    end

    local placeName = "Unknown Map"
    pcall(function()
        placeName = game:GetService("MarketplaceService"):GetProductInfo(game.PlaceId).Name
    end)

    -- โครงสร้าง JSON แบบยืดหยุ่นสูงสุด ส่งข้อมูลพื้นฐานพร้อมอาร์เรย์ของไอเท็มเฉพาะแมพ
    local body = HttpService:JSONEncode({
        key = cfg.Key,
        script_key = cfg.Key,
        userId = player.UserId,
        name = player.Name,
        displayName = player.DisplayName,
        coins = getCoins(),
        status = statusText or "Online",
        placeId = game.PlaceId,
        placeName = placeName,
        jobId = game.JobId,
        inventory = itemsArray or {},
        items = itemsArray or {} -- ส่งเป็น Array ลิสต์ไอเท็มไปให้หลังบ้านวนลูปเซฟลง player_inventory
    })

    task.spawn(function()
        local ok, result = pcall(function()
            return requestFunc({
                Url = cfg.Url,
                Method = "POST",
                Headers = {
                    ["Content-Type"] = "application/json",
                    ["X-Script-Key"] = cfg.Key or "",
                },
                Body = body
            })
        end)

        if ok then
            print("Dashboard response:", result and result.StatusCode, result and result.Body)
        else
            warn("Dashboard update failed:", result)
        end
    end)
end

-- ลูปทำงานหลัก (ตรวจสอบไอเท็มตามแมพทุกๆ 1 วินาที)
while task.wait(1) do
    local currentPlaceId = game.PlaceId
    local itemsToSend = {} -- ตารางสำหรับเก็บข้อมูลที่จะแปลงเป็น Dynamic Array ส่งไปหลังบ้าน

    -- ═════════════════════════════════════════════════════════════
    -- แมพที่ 1: Toilet Tower Defense (PlaceId: 13775256536)
    -- ═════════════════════════════════════════════════════════════
    if currentPlaceId == 13775256536 or currentPlaceId == 11654637731 then
        local UTC = 0
        local UTS = 0
        local TITAN = 0
        local Cenima = 0

        pcall(function()
            local lobby = player.PlayerGui:FindFirstChild("Lobby")
            local unitFrame = lobby and lobby:FindFirstChild("UnitFrame")
            local unitList = unitFrame and unitFrame:FindFirstChild("UnitList")

            if unitList then
                for _, row in pairs(unitList:GetChildren()) do
                    if string.sub(row.Name, 1, 3) == "Row" then
                        for _, desc in pairs(row:GetDescendants()) do
                            if desc.Name == "Price" and desc:IsA("TextLabel") then
                                local priceText = string.gsub(desc.Text, "%s+", "")

                                local troop = desc.Parent
                                local gradient = troop and troop:FindFirstChild("RarityGradient")

                                if gradient and gradient:IsA("UIGradient") then
                                    local kp = gradient.Color.Keypoints
                                    local r = kp[1].Value.R
                                    local g = kp[1].Value.G
                                    local b = kp[1].Value.B

                                    if r >= 0.95 and g <= 0.05 and b <= 0.05 then
                                        if priceText == "$1500" then
                                            UTC = UTC + 1
                                        elseif priceText == "$1000" then
                                            UTS = UTS + 1
                                        elseif priceText == "$2000" then
                                            Cenima = Cenima + 1
                                        end
                                    end
                                end
                            end
                        end
                    end
                end
            end
        end)

        -- จัดรูปแบบข้อมูล TTD ยัดใส่ Array (ระบุชื่อ item_id ให้ตรงกับที่คุณต้องการนำไปคิวรีหน้าเว็บ)
        table.insert(itemsToSend, { item_id = "UTC", amount = UTC })
        table.insert(itemsToSend, { item_id = "UTS", amount = UTS })
        table.insert(itemsToSend, { item_id = "TITAN", amount = TITAN })
        table.insert(itemsToSend, { item_id = "Cenima", amount = Cenima })

    -- ═════════════════════════════════════════════════════════════
    -- แมพที่ 2: Survive Zombie Arena (PlaceId: 114204398207377)
    -- ═════════════════════════════════════════════════════════════
    elseif currentPlaceId == 114204398207377 then
        local Credits = 0
        local VoidShards = 0
        local Class = "None"

        pcall(function()
            Credits = tonumber(player:FindFirstChild("Credits") and player.Credits.Value) or 0
            VoidShards = tonumber(player:FindFirstChild("VoidShards") and player.VoidShards.Value) or 0
            Class = player:FindFirstChild("SelectedClass") and player.SelectedClass.Value or "None"
        end)

        -- จัดรูปแบบข้อมูลแมพซอมบี้ ยัดใส่ Array
        -- (ถ้าค่าเป็นตัวเลขให้ส่งช่อง amount ถ้าเป็นข้อความให้ส่งช่อง text_value เพื่อให้สอดคล้องกับตาราง player_inventory)
        table.insert(itemsToSend, { item_id = "Credits", amount = Credits })
        table.insert(itemsToSend, { item_id = "VoidShards", amount = VoidShards })
        table.insert(itemsToSend, { item_id = "SelectedClass", text_value = Class })
    end

    -- ส่งข้อมูลไปอัปเดตที่ Dashboard หลังบ้าน
    SendDashboardUpdate("Online", itemsToSend)

    -- พิมพ์ Log เพื่อตรวจสอบการทำงานในหน้าจอ F9 Console ของเกม
    print("Sent Dynamic Dashboard Data successfully! Items sent count: " .. #itemsToSend)
end
